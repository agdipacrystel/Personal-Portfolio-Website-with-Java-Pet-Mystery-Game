Pet Mystery Game

This is my final performance task for CC102. It is a console-based Java guessing game where players identify different pets using clues.

Game Features

  • Three difficulty levels
  • Five questions per level
  • Pet guessing challenges
  • Three chances for each question
  • Score tracking
  • Play again option

Java Source Code


import java.util.Scanner;

public class PetMysteryGame {
    public static void main(String[] args) {

        Scanner e = new Scanner(System.in);

        String[][] pets = {
            {"cat", "dog", "rabbit", "fish", "bird"},
            {"owl", "penguin", "dolphin", "turtle", "fox"},
            {"platypus", "chameleon", "octopus", "hedgehog", "flamingo"}
        };

        String[][] clues = {
            {
                "I say meow and love to sleep.",
                "I bark and guard your home.",
                "I have long ears and love carrots.",
                "I live in water and have fins.",
                "I have wings and can fly."
            },
            {
                "I can turn my head almost all the way around.",
                "I am a bird but I cannot fly. I love cold places.",
                "I am a smart sea animal that uses sounds to communicate.",
                "I carry my home on my back.",
                "I have a bushy tail and I am known for being clever."
            },
            {
                "I am a mammal, but I lay eggs and have a duck-like bill.",
                "I can change my skin color to blend with my surroundings.",
                "I have eight arms and can squeeze through small spaces.",
                "I have spines on my back and curl into a ball for protection.",
                "I am a tall pink bird that often stands on one leg."
            }
        };

        String[] levels = {"Easy", "Intermediate", "Difficult"};

        String playAgain = "yes";

        System.out.println("==================================");
        System.out.println("        PET MYSTERY GAME");
        System.out.println("==================================");

        while (playAgain.equalsIgnoreCase("yes")) {

            int level = -1;
            int score = 0;

            System.out.println("\nSELECT DIFFICULTY");
            System.out.println("1. Easy");
            System.out.println("2. Intermediate");
            System.out.println("3. Difficult");

            while (level == -1) {

                System.out.print("\nEnter level: ");
                String choice = e.nextLine();

                if (choice.equalsIgnoreCase("easy") || choice.equals("1")) {
                    level = 0;
                } 
                else if (choice.equalsIgnoreCase("intermediate") || choice.equals("2")) {
                    level = 1;
                } 
                else if (choice.equalsIgnoreCase("difficult") || choice.equals("3")) {
                    level = 2;
                } 
                else {
                    System.out.println("Invalid choice! Try again.");
                }
            }

            System.out.println("\nYou selected " + levels[level] + " Mode!");
            System.out.println("You have 3 chances for each pet.");

            for (int i = 0; i < 5; i++) {

                int chances = 3;
                boolean correct = false;

                System.out.println("\n------------------------------");
                System.out.println("MYSTERY PET #" + (i + 1));
                System.out.println("Clue: " + clues[level][i]);

                while (chances > 0) {

                    System.out.print("Your answer: ");
                    String answer = e.nextLine();

                    if (answer.equalsIgnoreCase(pets[level][i])) {

                        System.out.println("Correct! Amazing!");
                        score++;
                        correct = true;
                        break;

                    } else {

                        chances--;
                        System.out.println("Wrong answer!");

                        if (chances > 0) {
                            System.out.println("Chances left: " + chances);
                        }
                    }
                }

                if (!correct) {
                    System.out.println("The correct answer is: " + pets[level][i]);
                }
            }

            System.out.println("\n==================================");
            System.out.println("             GAME OVER");
            System.out.println("==================================");

            System.out.println("Difficulty: " + levels[level]);
            System.out.println("Your Score: " + score + "/5");

            if (score == 5) {
                System.out.println("Perfect! You are a Pet Master!");
            } 
            else if (score >= 3) {
                System.out.println("Great job, Pet Detective!");
            } 
            else {
                System.out.println("Keep practicing, Detective!");
            }

            System.out.print("\nDo you want to play again? (yes/no): ");
            playAgain = e.nextLine();

            while (!playAgain.equalsIgnoreCase("yes")
                    && !playAgain.equalsIgnoreCase("no")) {

                System.out.print("Please enter yes or no: ");
                playAgain = e.nextLine();
            }
        }

        System.out.println("\nThank you for playing!");
        System.out.println("Goodbye, Pet Detective!");

        e.close();
    }
}
